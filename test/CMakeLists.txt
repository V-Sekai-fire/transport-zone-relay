# Witness-cpp properties of src/otel_span.h, of the caps in src/zone_caps.c, of the state machine in src/zone_relay.c and of the
# relay on sockets in src/zone_net.c. A planted build runs the same properties on a copy of the source with one defect written in. Three more builds plant a defect each (the end time written as the start; the parent id left out; the status left out); the same properties must fail on them, which is how the check proves it can fail.
cmake_minimum_required(VERSION 3.16)
project(otel-span-witness LANGUAGES C CXX)
include(FetchContent)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5 CACHE STRING "" FORCE)
FetchContent_Declare(witness GIT_REPOSITORY https://github.com/V-Sekai-fire/contract-witness-cpp.git GIT_TAG main GIT_SHALLOW TRUE)
set(WITNESS_BUILD_TESTS OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(witness)
FetchContent_Declare(doctest GIT_REPOSITORY https://github.com/doctest/doctest.git GIT_TAG v2.4.11 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(doctest)
enable_testing()
foreach(variant IN ITEMS real planted_pbtimes planted_pbparent planted_pbstatus)
  add_executable(otel_span_${variant} otel_span_witness.cpp)
  target_link_libraries(otel_span_${variant} PRIVATE witness::witness doctest::doctest)
  target_compile_options(otel_span_${variant} PRIVATE -Wall -Wextra -Werror)
  target_compile_definitions(otel_span_${variant} PRIVATE DOCTEST_CONFIG_USE_STD_HEADERS)
endforeach()
target_compile_definitions(otel_span_planted_pbtimes PRIVATE OTEL_PLANT_PBTIMES)
target_compile_definitions(otel_span_planted_pbparent PRIVATE OTEL_PLANT_PBPARENT)
target_compile_definitions(otel_span_planted_pbstatus PRIVATE OTEL_PLANT_PBSTATUS)
add_test(NAME otel_span_real COMMAND otel_span_real)
add_test(NAME otel_span_planted_pbtimes COMMAND otel_span_planted_pbtimes)
add_test(NAME otel_span_planted_pbparent COMMAND otel_span_planted_pbparent)
add_test(NAME otel_span_planted_pbstatus COMMAND otel_span_planted_pbstatus)
set_tests_properties(otel_span_planted_pbtimes otel_span_planted_pbparent otel_span_planted_pbstatus PROPERTIES WILL_FAIL TRUE)

# The caps. The real build runs the properties on src/zone_caps.c. Each planted build runs them on a copy of it with one
# defect written in at configure time, and its test must fail.
add_executable(zone_caps_real zone_caps_witness.cpp ../src/zone_caps.c)
target_link_libraries(zone_caps_real PRIVATE witness::witness doctest::doctest)
target_compile_options(zone_caps_real PRIVATE -Wall -Wextra -Werror)
target_compile_definitions(zone_caps_real PRIVATE DOCTEST_CONFIG_USE_STD_HEADERS)

file(READ ${CMAKE_CURRENT_SOURCE_DIR}/../src/zone_caps.c ZC_SOURCE)
function(plant name from to)
  string(REPLACE "${from}" "${to}" planted "${ZC_SOURCE}")
  if(planted STREQUAL ZC_SOURCE)
    message(FATAL_ERROR "the planted ${name} was not planted")
  endif()
  file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/planted/zone_caps_${name}.c "${planted}")
  add_executable(zone_caps_planted_${name} zone_caps_witness.cpp ${CMAKE_CURRENT_BINARY_DIR}/planted/zone_caps_${name}.c)
  target_include_directories(zone_caps_planted_${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../src)
  target_link_libraries(zone_caps_planted_${name} PRIVATE witness::witness doctest::doctest)
  target_compile_options(zone_caps_planted_${name} PRIVATE -Wall -Wextra -Werror)
  target_compile_definitions(zone_caps_planted_${name} PRIVATE DOCTEST_CONFIG_USE_STD_HEADERS)
  add_test(NAME zone_caps_planted_${name} COMMAND zone_caps_planted_${name})
  set_tests_properties(zone_caps_planted_${name} PROPERTIES WILL_FAIL TRUE)
endfunction()
