# Witness-cpp properties of src/otel_span.h. Two more builds plant a defect each (no string cleaning; the end time written as the start); the same properties must fail on them, which is how the check proves it can fail.
cmake_minimum_required(VERSION 3.16)
project(otel-span-witness LANGUAGES C CXX)
include(FetchContent)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5 CACHE STRING "" FORCE)
FetchContent_Declare(witness GIT_REPOSITORY https://github.com/V-Sekai-fire/contract-witness-cpp.git GIT_TAG main GIT_SHALLOW TRUE)
set(WITNESS_BUILD_TESTS OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(witness)
FetchContent_Declare(doctest GIT_REPOSITORY https://github.com/doctest/doctest.git GIT_TAG v2.4.11 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(doctest)
enable_testing()
foreach(variant IN ITEMS real planted planted_pbtimes)
  add_executable(otel_span_${variant} otel_span_witness.cpp)
  target_link_libraries(otel_span_${variant} PRIVATE witness::witness doctest::doctest)
  target_compile_options(otel_span_${variant} PRIVATE -Wall -Wextra -Werror)
  target_compile_definitions(otel_span_${variant} PRIVATE DOCTEST_CONFIG_USE_STD_HEADERS)
endforeach()
target_compile_definitions(otel_span_planted PRIVATE OTEL_PLANT_NOCLEAN)
target_compile_definitions(otel_span_planted_pbtimes PRIVATE OTEL_PLANT_PBTIMES)
add_test(NAME otel_span_real COMMAND otel_span_real)
add_test(NAME otel_span_planted_nocleaning COMMAND otel_span_planted)
add_test(NAME otel_span_planted_pbtimes COMMAND otel_span_planted_pbtimes)
set_tests_properties(otel_span_planted_nocleaning otel_span_planted_pbtimes PROPERTIES WILL_FAIL TRUE)
